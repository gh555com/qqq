function logMessage(message, level = "WARN") {
    if (level !== "ERROR" && level !== "WARN") return;
    const ts = new Date().toISOString();
    const line = `[${ts}] [${level}] ${message}\n`;
    try {
        fs.mkdirSync(path.dirname(LOG_PATH), { recursive: true });
        fs.appendFileSync(LOG_PATH, line);
    } catch (e) {
        console.error("鏃ュ織鍐欏叆澶辫触:", e);
    }
}

function runPythonScript(additionalEnv = {}) {
    const scriptPath = path.join(__dirname, "kp.py");
    if (!fs.existsSync(scriptPath)) {
        logMessage("鑴氭湰涓嶅瓨鍦? " + scriptPath, "ERROR");
        vscode.window.showErrorMessage("鑴氭湰涓嶅瓨鍦?);
        return;
    }

    // 鍚堝苟鍩虹鐜鍙橀噺鍜岄澶栫幆澧冨彉閲?    const env = {
        ...process.env,
        PYTHONIOENCODING: "utf-8",
        ...additionalEnv,
    };

    const child = cp.spawn("python", [scriptPath], {
        stdio: ["pipe", "pipe", "pipe"],
        env: env,
    });

    let stdout = "";
    let stderr = "";
    // 鏄庣‘鎸囧畾 UTF-8 缂栫爜
    child.stdout.on("data", (d) => (stdout += d.toString("utf8")));
    child.stderr.on("data", (d) => (stderr += d.toString("utf8")));

    child.on("close", (code) => {
        if (stderr) logMessage("Python stderr: " + stderr, "WARN");
        if (code !== 0) {
            logMessage("閫€鍑虹爜 " + code, "ERROR");
            vscode.window.showErrorMessage("鎵ц澶辫触 " + code);
            return;
        }
        try {
            const result = JSON.parse(stdout.trim());
            handleResult(result);
        } catch (e) {
            logMessage("JSON瑙ｆ瀽澶辫触: " + e.message, "ERROR");
        }
    });
}

function handleResult(result) {
    if (result.error) {
        logMessage("澶勭悊澶辫触: " + result.error, "ERROR");
        vscode.window.showErrorMessage(result.error);
        return;
    }
    const ed = vscode.window.activeTextEditor;
    switch (result.type) {
        case "folder_text":
            // 澶勭悊鏂囦欢澶硅矾寰勶紝鐩存帴鎻掑叆鏂囨湰锛屼笉鍋氫换浣曚慨楗?            if (ed) {
                ed.edit((edit) => edit.insert(ed.selection.active, result.text));
            }
            break;
        case "text":
            if (ed) {
                ed.edit((edit) => edit.insert(ed.selection.active, result.text)).then(
                    () => {
                        setTimeout(() => renderImages(ed), 50);
                    },
                );
            }
            break;
        case "image":
            if (ed) {
                ed.edit((edit) =>
                    edit.insert(ed.selection.active, `[${result.path}]`),
                ).then(() => {
                    setTimeout(() => renderImages(ed), 50);
                });
            }
            break;
        case "file":
            if (result.files && result.files.length > 0) {
                if (ed) {
                    // 鎻掑叆鎵€鏈夋枃浠剁殑鏍囪瘑绗︼紝鏃犺浠€涔堟牸寮?                    const fileMarkers = result.files.map((f) => `[${f}]`).join("\n");
                    ed.edit((edit) => edit.insert(ed.selection.active, fileMarkers)).then(
                        () => {
                            setTimeout(() => renderImages(ed), 50);
                        },
                    );
                }
                vscode.window.showInformationMessage(
                    "鏂囦欢宸插鍒?" + result.files.length,
                );
            }
            break;
        case "binary":
            vscode.window.showInformationMessage(
                "浜岃繘鍒跺凡淇濆瓨 " + path.basename(result.path),
            );
            break;
        case "cancelled":
            // 鐢ㄦ埛鍙栨秷浜嗙矘璐存搷浣?            vscode.window.showInformationMessage("绮樿创宸插彇娑?);
            break;
        default:
            vscode.window.showWarningMessage("鏈煡鍐呭");
    }
}

class FileCodeLensProvider {
    provideCodeLenses(document) {
        const lenses = [];
        const regex = /\[([A-Za-z]:[\\\/]view[\\\/]p[\\\/][^\[\]]+)\]/gi;
        const text = document.getText();
        let match;

        while ((match = regex.exec(text))) {
            const pos = document.positionAt(match.index);
            const range = new vscode.Range(pos, pos);
            const absPath = match[1].replace(/\//g, "\\");

            // 妫€鏌ユ枃浠舵槸鍚﹀瓨鍦?            if (fs.existsSync(absPath)) {
                lenses.push(
                    new vscode.CodeLens(range, {
                        title: blockMode ? "qqq" : "aaa",
                        command: "qqq.openFile",
                        arguments: [absPath],
                    }),
                );
            }
        }
        return lenses;
    }
}

function debounceRender(editor, delay = 100) {
    // 濡傛灉姝ｅ湪鎵ц娓叉煋锛屽拷鐣ユ柊鐨勮Е鍙?    if (debounceRender.isProcessing) {
        return;
    }

    clearTimeout(debounceRender.timer);
    debounceRender.timer = setTimeout(() => {
        if (editor && !editor.document.isClosed) {
            // 鏍囪涓烘鍦ㄥ鐞?            debounceRender.isProcessing = true;

            // 鎵ц娓叉煋锛屽苟鍦ㄥ畬鎴愬悗鍙栨秷澶勭悊鏍囪
            Promise.resolve()
                .then(() => {
                    renderImages(editor);
                })
                .finally(() => {
                    // 纭繚鏃犺鎴愬姛澶辫触閮戒細鍙栨秷澶勭悊鏍囪
                    setTimeout(() => {
                        debounceRender.isProcessing = false;
                    }, 50); // 灏忓欢杩熺‘淇濇覆鏌撳畬鎴?                });
        }
    }, delay);
}

// 鍒濆鍖栧鐞嗙姸鎬佹爣蹇?debounceRender.isProcessing = false;

/**
 * 閫掑綊璁＄畻鏂囦欢澶规€诲ぇ灏? * @param {string} dirPath
 * @returns {Promise<number>} 鏂囦欢澶规€诲ぇ灏?(瀛楄妭)
 */
function calculateFolderSizeRecursive(dirPath) {
    return new Promise((resolve, reject) => {
        let totalSize = 0;

        function readDir(currentDir) {
            try {
                const entries = fs.readdirSync(currentDir, { withFileTypes: true });

                for (const entry of entries) {
                    const entryPath = path.join(currentDir, entry.name);

                    try {
                        const stat = fs.statSync(entryPath);

                        if (entry.isDirectory()) {
                            // 閫掑綊璋冪敤
                            totalSize += readDir(entryPath);
                        } else {
                            // 鏂囦欢灏哄
                            totalSize += stat.size;
                        }
                    } catch (error) {
                        // 蹇界暐鏃犳硶璁块棶鐨勬枃浠?鐩綍锛岀户缁?                    }
                }
                return totalSize;
            } catch (error) {
                // 濡傛灉鐩綍鏃犳硶璇诲彇锛岃繑鍥炲綋鍓嶅凡绱鐨勫ぇ灏?                logMessage(`璇诲彇鐩綍澶辫触: ${currentDir}`, "WARN");
                return totalSize;
            }
        }

        // 寮傛鎵ц璁＄畻
        const worker = new Promise((res) => {
            setTimeout(() => {
                try {
                    const size = readDir(dirPath);
                    res(size);
                } catch (e) {
                    logMessage(`寮傛璁＄畻鏂囦欢澶瑰ぇ灏忓け璐? ${e.message}`, "ERROR");
                    res(0);
                }
            }, 0);
        });

        worker
            .then((size) => {
                resolve(size);
            })
            .catch(reject);
    });
}

/**
 * 鏍煎紡鍖栨枃浠跺ぇ灏忥紝鍥涜垗浜斿叆鍙栨暣
 * @param {number} size 瀛楄妭澶у皬
 * @param {string} unitSizeMode 'b', 'k', 'm' 涓殑涓€涓? * @returns {{size: number, unit: string}} 鏍煎紡鍖栧悗鐨勫ぇ灏忓拰鍗曚綅
 */
function formatFileSize(size, unitSizeMode) {
    const ONE_K = 1024;
    const ONE_M = 1024 * 1024;
    const ONE_B = 1;

    let targetUnit;
    let targetDivisor;
    let maxUnit = "m";

    // 纭畾鐩爣鍗曚綅鍜岄櫎鏁?    if (
        unitSizeMode === "m" ||
        (unitSizeMode === "k" && size >= ONE_M) ||
        (unitSizeMode === "b" && size >= ONE_M)
    ) {
        targetUnit = "m";
        targetDivisor = ONE_M;
    } else if (unitSizeMode === "k" || (unitSizeMode === "b" && size >= ONE_K)) {
        targetUnit = "k";
        targetDivisor = ONE_K;
    } else {
        targetUnit = "b";
